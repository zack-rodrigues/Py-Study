16. Equação do Segundo Grau (Bhaskara)

Calcule as raízes de uma equação ax² + bx + c = 0. Faça todas as validações: se a=0 (não é 2º grau), se delta < 0 (não possui raízes reais), delta = 0 (1 raiz real) ou delta > 0 (2 raízes).

import math

print("Equação: ax² + bx + c = 0")
a = float(input("Valor de a: "))

if a == 0:
    print("Se 'a' é igual a zero, a equação não é do 2º grau. Programa encerrado.")
else:
    b = float(input("Valor de b: "))
    c = float(input("Valor de c: "))

    delta = (b ** 2) - (4 * a * c)

    if delta < 0:
        print(f"Delta ({delta}) é negativo. A equação não possui raízes reais.")
    elif delta == 0:
        raiz = -b / (2 * a)
        print(f"Delta é zero. A equação possui uma raiz real: {raiz:.2f}")
    else:
        raiz1 = (-b + math.sqrt(delta)) / (2 * a)
        raiz2 = (-b - math.sqrt(delta)) / (2 * a)
        print(f"Delta é {delta}. As raízes são: {raiz1:.2f} e {raiz2:.2f}")
