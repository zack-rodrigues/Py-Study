3. Classe Retângulo

Atributos: Lado A (base), Lado B (altura)

Métodos: mudarLados(), retornarLados(), calcularArea(), calcularPerimetro()

Extra: Crie um programa que utilize esta classe para calcular a quantidade de pisos e rodapés necessários para um local com medidas fornecidas pelo usuário.

class Retangulo:
    def __init__(self, base, altura):
        self.base = base
        self.altura = altura
        
    def mudarLados(self, nova_base, nova_altura):
        self.base = nova_base
        self.altura = nova_altura
        
    def retornarLados(self):
        return self.base, self.altura
        
    def calcularArea(self):
        return self.base * self.altura
        
    def calcularPerimetro(self):
        return 2 * (self.base + self.altura)

# Programa Extra
print("--- Cálculo de Materiais ---")
base_local = float(input("Informe a base do local (m): "))
altura_local = float(input("Informe a altura do local (m): "))

local = Retangulo(base_local, altura_local)

area = local.calcularArea()
perimetro = local.calcularPerimetro()

print(f"\nSerão necessários {area} m² de piso.")
print(f"Serão necessários {perimetro} metros de rodapé.")

4. Classe Pessoa

Atributos: nome, idade, peso, altura

Métodos: envelhecer(), engordar(), emagrecer(), crescer()

Regra: Por padrão, a cada ano que nossa pessoa envelhece (até os 21 anos), ela deve crescer 0,5 cm.

class Pessoa:
    def __init__(self, nome, idade, peso, altura):
        self.nome = nome
        self.idade = idade
        self.peso = peso
        self.altura = altura # em metros, ex: 1.70
        
    def envelhecer(self, anos=1):
        for _ in range(anos):
            self.idade += 1
            if self.idade < 21:
                self.crescer(0.005) # 0.5 cm
                
    def engordar(self, kg):
        self.peso += kg
        
    def emagrecer(self, kg):
        self.peso -= kg
        
    def crescer(self, metros):
        self.altura += metros
        
    def __str__(self):
        return f"{self.nome}: {self.idade} anos, {self.peso}kg, {self.altura:.2f}m"

# Testando
joao = Pessoa("João", 18, 70.0, 1.75)
print(joao)
joao.envelhecer(3) # Vai para 21 anos e cresce 1 cm (0.5 aos 19, 0.5 aos 20)
print(joao)

5. Classe Conta Corrente

Atributos: número da conta, nome do correntista e saldo (default = 0).

Métodos: alterarNome(), depositar(), sacar()

class ContaCorrente:
    # O parâmetro saldo=0 define o valor padrão caso o usuário não informe
    def __init__(self, numero_conta, nome_correntista, saldo=0.0):
        self.numero_conta = numero_conta
        self.nome_correntista = nome_correntista
        self.saldo = saldo
        
    def alterarNome(self, novo_nome):
        self.nome_correntista = novo_nome
        
    def depositar(self, valor):
        if valor > 0:
            self.saldo += valor
            
    def sacar(self, valor):
        if 0 < valor <= self.saldo:
            self.saldo -= valor
        else:
            print("Saldo insuficiente ou valor inválido.")

# Testando
conta = ContaCorrente("1234-5", "Ana Silva")
conta.depositar(500)
conta.sacar(150)
print(f"Titular: {conta.nome_correntista} | Saldo: R$ {conta.saldo:.2f}")

7. Classe Bichinho Virtual (Tamagushi)

Atributos: nome, fome, saude, idade

Métodos: alterarNome(), alterarFome(), alterarSaude(), alterarIdade(), retornarHumor()

Regra: O humor é uma combinação entre os atributos Fome e Saúde (ex: a média dos dois).

class Tamagushi:
    def __init__(self, nome):
        self.nome = nome
        self.fome = 100   # 100 = Sem fome
        self.saude = 100  # 100 = Muito saudável
        self.idade = 1
        
    def alterarNome(self, novo_nome): self.nome = novo_nome
    def alterarFome(self, valor): self.fome = valor
    def alterarSaude(self, valor): self.saude = valor
    def alterarIdade(self, valor): self.idade = valor
        
    def retornarHumor(self):
        # Humor é a média entre saúde e saciedade
        humor = (self.fome + self.saude) / 2
        if humor >= 70: return "Feliz"
        elif humor >= 40: return "Neutro"
        else: return "Triste/Doente"

# Testando
pet = Tamagushi("Rex")
print(f"Humor inicial: {pet.retornarHumor()}")
pet.alterarFome(30)
pet.alterarSaude(40)
print(f"Humor após ficar doente e com fome: {pet.retornarHumor()}")

8. Classe Macaco

Atributos: nome, bucho (uma lista)

Métodos: comer(alimento), verBucho(), digerir()

Extra: Teste criar dois macacos, alimentá-los, e tente fazer um macaco comer o outro (macaco canibal).

class Macaco:
    def __init__(self, nome):
        self.nome = nome
        self.bucho = []
        
    def comer(self, alimento):
        # Adiciona o alimento ao bucho
        self.bucho.append(alimento)
        
    def verBucho(self):
        print(f"Bucho do {self.nome}: {self.bucho}")
        
    def digerir(self):
        if self.bucho:
            digerido = self.bucho.pop(0) # Remove o primeiro que entrou
            print(f"{self.nome} digeriu {digerido}.")
        else:
            print(f"O bucho de {self.nome} está vazio.")
            
    # O método __str__ diz como o Python deve exibir o objeto quando usarmos print()
    def __repr__(self):
        return f"Macaco({self.nome})"

# Testando
george = Macaco("George")
kong = Macaco("Kong")

george.comer("Banana")
george.comer("Maçã")
george.verBucho()
george.digerir()

kong.comer("Melancia")
kong.comer(george) # Kong comendo o George!
kong.verBucho()

9. Classe Ponto e Retângulo

  • Classe Ponto: Atributos x, y.
  • Classe Retângulo: Atributos largura, altura e um Ponto inicial (representando o canto inferior esquerdo).
  • Métodos: Função para encontrar o centro do retângulo (retorna um novo Ponto).
class Ponto:
    def __init__(self, x=0, y=0):
        self.x = x
        self.y = y
        
    def imprimir(self):
        print(f"Ponto({self.x}, {self.y})")

class Retangulo:
    def __init__(self, largura, altura, ponto_origem):
        self.largura = largura
        self.altura = altura
        self.origem = ponto_origem # Este atributo recebe um objeto da classe Ponto
        
    def encontrarCentro(self):
        # O centro no eixo X é a origem X + metade da largura
        centro_x = self.origem.x + (self.largura / 2)
        # O centro no eixo Y é a origem Y + metade da altura
        centro_y = self.origem.y + (self.altura / 2)
        
        # Retorna um NOVO objeto Ponto
        return Ponto(centro_x, centro_y)

# Testando com um menu simples
p_origem = Ponto(10, 10)
meu_retangulo = Retangulo(100, 50, p_origem)

while True:
    print("\n1. Encontrar e imprimir Centro")
    print("2. Alterar dimensões do Retângulo")
    print("3. Sair")
    opc = input("Escolha: ")
    
    if opc == '1':
        centro = meu_retangulo.encontrarCentro()
        print("-> Centro do Retângulo: ", end="")
        centro.imprimir()
    elif opc == '2':
        meu_retangulo.largura = float(input("Nova Largura: "))
        meu_retangulo.altura = float(input("Nova Altura: "))
    elif opc == '3':
        break

10. Classe Bomba de Combustível

Atributos: tipoCombustivel, valorLitro, quantidadeCombustivel (estoque da bomba)

Métodos: abastecerPorValor(), abastecerPorLitro(), alterarValor(), alterarCombustivel(), alterarQuantidadeCombustivel()

class BombaCombustivel:
    def __init__(self, tipo, valor, quantidade):
        self.tipoCombustivel = tipo
        self.valorLitro = valor
        self.quantidadeCombustivel = quantidade # Estoque da bomba
        
    def abastecerPorValor(self, valor):
        litros = valor / self.valorLitro
        if litros <= self.quantidadeCombustivel:
            self.quantidadeCombustivel -= litros
            print(f"Abastecido: {litros:.2f} litros. Total a pagar: R$ {valor:.2f}")
        else:
            print("Não há combustível suficiente na bomba.")
            
    def abastecerPorLitro(self, litros):
        if litros <= self.quantidadeCombustivel:
            valor = litros * self.valorLitro
            self.quantidadeCombustivel -= litros
            print(f"Abastecido: {litros:.2f} litros. Total a pagar: R$ {valor:.2f}")
        else:
            print("Não há combustível suficiente na bomba.")
            
    def alterarValor(self, novo_valor): self.valorLitro = novo_valor
    def alterarCombustivel(self, novo_tipo): self.tipoCombustivel = novo_tipo
    def alterarQuantidadeCombustivel(self, nova_qtde): self.quantidadeCombustivel = nova_qtde

# Testando
bomba = BombaCombustivel("Gasolina", 5.50, 1000)
bomba.abastecerPorValor(50.0) # Abastece 50 reais
bomba.abastecerPorLitro(10)   # Abastece 10 litros
print(f"Estoque restante: {bomba.quantidadeCombustivel:.2f} litros")

11. Classe Carro

Atributos: consumo (km/litro), combustível (inicia sempre em 0)

Métodos: andar(distancia), obterGasolina(), adicionarGasolina(qtde)

class Carro:
    def __init__(self, consumo_km_litro):
        self.consumo = consumo_km_litro
        self.combustivel = 0.0
        
    def adicionarGasolina(self, litros):
        self.combustivel += litros
        
    def andar(self, distancia):
        litros_necessarios = distancia / self.consumo
        
        if self.combustivel >= litros_necessarios:
            self.combustivel -= litros_necessarios
            print(f"O carro andou {distancia}km.")
        else:
            print("Gasolina insuficiente para o trajeto.")
            
    def obterGasolina(self):
        return self.combustivel

# Testando
meuFusca = Carro(15) # 15 km por litro
meuFusca.adicionarGasolina(20) # Abastece 20 litros
meuFusca.andar(100) # Anda 100 km
print(f"Gasolina restante: {meuFusca.obterGasolina():.2f} litros")

12. Classe Conta de Investimento

Atributos: saldo, taxaJuros

Métodos: adicioneJuros() (aplica a taxa de juros sobre o saldo atual)

Teste: Inicie com R$1000, taxa 10%, e aplique o método 5 vezes consecutivas para ver os juros compostos.

class ContaInvestimento:
    def __init__(self, saldo_inicial, taxa_juros):
        self.saldo = saldo_inicial
        self.taxa_juros = taxa_juros # ex: 10 para 10%
        
    def adicioneJuros(self):
        # Aumenta o saldo baseado na porcentagem de juros
        self.saldo += self.saldo * (self.taxa_juros / 100)
        
    def getSaldo(self):
        return self.saldo

# Testando Juros Compostos
conta = ContaInvestimento(1000.0, 10.0)

for i in range(5):
    conta.adicioneJuros()
    print(f"Mês {i+1}: R$ {conta.getSaldo():.2f}")

14. Bichinho Virtual++ (Fazenda)

Aprimore a classe Tamagushi (Bichinho Virtual):

  • Permita receber parâmetros exatos para "alimentar" (quantidade) e "brincar" (tempo).
  • Adicione o método __str__ que funciona como uma porta secreta (mostra status interno).
  • Crie uma "Fazenda" controlando uma lista de objetos Bichinho (Alimente e brinque com todos de uma vez!).
class BichinhoPlus:
    def __init__(self, nome):
        self.nome = nome
        self.fome = 50
        self.tedio = 50
        
    def alimentar(self, quantidade):
        self.fome -= quantidade
        if self.fome < 0: self.fome = 0
            
    def brincar(self, tempo):
        self.tedio -= tempo
        if self.tedio < 0: self.tedio = 0
            
    def ouvir(self):
        print(f"{self.nome} diz: Minha fome é {self.fome} e meu tédio é {self.tedio}.")
        
    # A Porta Secreta!
    def __str__(self):
        return f"[{self.nome} - Atributos Ocultos: Fome:{self.fome} Tedio:{self.tedio}]"

# Fazenda de Bichinhos (Lista de Objetos)
fazenda = [
    BichinhoPlus("Vaca"),
    BichinhoPlus("Porco"),
    BichinhoPlus("Galinha")
]

print("--- Interagindo com a Fazenda Inteira ---")

# Alimentando todos de uma vez (Polimorfismo / Loop)
for animal in fazenda:
    animal.alimentar(20)
    animal.brincar(10)
    animal.ouvir()

print("\n--- Usando a Porta Secreta ---")
for animal in fazenda:
    print(animal) # Isso chama o método __str__ de cada objeto
