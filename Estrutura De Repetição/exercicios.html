divida_inicial = float(input("Valor da dívida: R$ "))

print("\nValor da Dívida | Juros | Quantidade de Parcelas | Valor da Parcela")

parcelas_juros = [(1, 0), (3, 10), (6, 15), (9, 20), (12, 25)]

for parcelas, juros_porcento in parcelas_juros:
    valor_juros = divida_inicial * (juros_porcento / 100)
    divida_total = divida_inicial + valor_juros
    valor_parcela = divida_total / parcelas
    
    print(f"R$ {divida_total:<12.2f} | R$ {valor_juros:<4.2f} | {parcelas:<22} | R$ {valor_parcela:.2f}")
