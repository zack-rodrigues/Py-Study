1. Controle de Cotas de Disco

A ACME Inc., uma organização com mais de 1500 funcionários, está tendo problemas de espaço em disco no seu servidor de arquivos. Você foi encarregado de desenvolver um programa para gerar o relatório gerencial de uso.

📅 Arquivo de entrada: usuarios.txt
alexandre       456123789
anderson        1245698456
antonio         123456456
carlos          91257581
cesar           987458
rosemary        789456125
Desafios Adicionais (Aplicados na Resposta):
  • Ordenar os usuários pelo percentual de espaço ocupado (do maior para o menor).
  • Gerar a saída final não apenas no console, mas em uma página HTML.
import os

def bytes_para_mb(bytes_val):
    return bytes_val / (1024 * 1024)

# 1. Gerando o arquivo de teste (Mock)
with open("usuarios.txt", "w") as f:
    f.write("alexandre       456123789\nanderson        1245698456\n"
            "antonio         123456456\ncarlos          91257581\n"
            "cesar           987458\nrosemary        789456125\n")

usuarios = []
total_bytes = 0

# 2. Leitura e processamento de dados
with open("usuarios.txt", "r") as f:
    for linha in f:
        partes = linha.split()
        if len(partes) == 2:
            nome = partes[0]
            espaco = int(partes[1])
            # Armazenando como Dicionário para facilitar a ordenação
            usuarios.append({"nome": nome, "espaco": espaco})
            total_bytes += espaco

# 3. Desafio Extra: Ordenar do maior para o menor consumo de disco
usuarios.sort(key=lambda x: x["espaco"], reverse=True)

total_mb = bytes_para_mb(total_bytes)
media_mb = total_mb / len(usuarios) if usuarios else 0

# 4. Desafio Extra: Gerando o relatório em HTML
html = f"""



    
    Relatório ACME Inc.
    


    
ACME Inc. - Uso do espaço em disco pelos usuários

    
"""

for i, u in enumerate(usuarios):
    espaco_mb = bytes_para_mb(u["espaco"])
    perc = (u["espaco"] / total_bytes) * 100
    html += f"\n"

html += f"""
    

        
Nr.UsuárioEspaço (MB)% do uso
{i+1}{u['nome']}{espaco_mb:.2f} MB{perc:.2f}%

    

        
Espaço total ocupado: {total_mb:.2f} MB

        
Espaço médio ocupado: {media_mb:.2f} MB

    



"""

# Salvando a página
with open("relatorio_acme.html", "w", encoding="utf-8") as f:
    f.write(html)

print("✅ Análise concluída! O arquivo 'relatorio_acme.html' foi gerado no diretório atual com a listagem ordenada.")

2. Analisador de Logs do Apache

Desenvolva um analisador de log do Apache HTTP Server que varra o arquivo e extraia as strings de pesquisa do Google que estão redirecionando internautas para o site da sua organização. Ao final, conte e mostre o ranking das mais pesquisadas.

import urllib.parse
from collections import Counter

# 1. Gerando um Log de servidor (Mock) para testar o código
log_mock = """
127.0.0.1 - - [10/Jul/2026:12:00:00] "GET / HTTP/1.1" 200 "https://www.google.com/search?q=tutorial+de+python" "Mozilla/5.0"
127.0.0.1 - - [10/Jul/2026:12:05:00] "GET / HTTP/1.1" 200 "https://www.google.com/search?q=exercicios+de+programacao" "Mozilla/5.0"
127.0.0.1 - - [10/Jul/2026:12:10:00] "GET / HTTP/1.1" 200 "https://www.google.com/search?q=tutorial+de+python" "Mozilla/5.0"
127.0.0.1 - - [10/Jul/2026:12:15:00] "GET / HTTP/1.1" 200 "https://www.bing.com/search?q=isso+sera+ignorado" "Mozilla/5.0"
127.0.0.1 - - [10/Jul/2026:12:20:00] "GET / HTTP/1.1" 200 "https://www.google.com/search?q=como+analisar+logs" "Mozilla/5.0"
"""
with open("apache_access.log", "w") as f:
    f.write(log_mock)

# 2. O Programa Real
termos_pesquisa = []

with open("apache_access.log", "r") as f:
    for linha in f:
        # Verifica se a requisição veio do Google
        if "google.com" in linha:
            try:
                # Extraindo o link do Referrer (ficam entre aspas duplas nos logs do Apache)
                partes = linha.split('"')
                referrer = partes[3] 
                
                # urllib facilita o parse da URL e da Query String (tudo depois do '?')
                parsed_url = urllib.parse.urlparse(referrer)
                query_params = urllib.parse.parse_qs(parsed_url.query)
                
                # A pesquisa do Google sempre fica na variável 'q'
                if 'q' in query_params:
                    # parse_qs retorna uma lista, pegamos o índice 0
                    termo = query_params['q'][0] 
                    termos_pesquisa.append(termo)
            except Exception as e:
                continue

# Counter já faz o trabalho de agrupar e contar incidências repetidas!
contagem = Counter(termos_pesquisa)

print("🔎 TOP Termos Pesquisados no Google que levaram ao nosso site:\n")
for termo, freq in contagem.most_common(5):
    print(f"-> '{termo}': {freq} acesso(s)")

3. Analisador de Logs do Squid (Firewall)

Desenvolva um analisador de log do Squid Proxy/Firewall que identifique a flag TCP_DENIED/403. O programa deve processar as URLs barradas, extrair apenas o domínio principal, e exibir um ranking de quais são os sites mais bloqueados na rede da organização.

import urllib.parse
from collections import Counter

# 1. Gerando um Log do Squid (Mock) para testar o código
log_mock = """
1678888888.1 100 192.168.0.10 TCP_DENIED/403 0 GET http://www.facebook.com/videos - HIER_NONE/- text/html
1678888889.1 100 192.168.0.11 TCP_DENIED/403 0 GET http://www.youtube.com/watch - HIER_NONE/- text/html
1678888890.1 100 192.168.0.10 TCP_DENIED/403 0 GET http://www.facebook.com/messages - HIER_NONE/- text/html
1678888891.1 100 192.168.0.12 TCP_MISS/200 1024 GET http://www.python.org/ - DIRECT/1.2.3.4 text/html
1678888892.1 100 192.168.0.15 TCP_DENIED/403 0 GET http://www.instagram.com/ - HIER_NONE/- text/html
1678888893.1 100 192.168.0.10 TCP_DENIED/403 0 GET http://www.youtube.com/shorts - HIER_NONE/- text/html
"""
with open("squid.log", "w") as f:
    f.write(log_mock)

# 2. O Programa Real
sites_bloqueados = []

with open("squid.log", "r") as f:
    for linha in f:
        partes = linha.split()
        
        # O Squid divide os campos por espaço. O Status da requisição fica no índice 3.
        # Filtramos apenas as requisições barradas pelo firewall (TCP_DENIED)
        if len(partes) > 6 and "TCP_DENIED" in partes[3]:
            url = partes[6] # A URL inteira acessada fica no índice 6
            
            try:
                # .netloc puxa apenas o domínio (ex: www.facebook.com), removendo "/videos" e derivados.
                dominio = urllib.parse.urlparse(url).netloc
                sites_bloqueados.append(dominio)
            except Exception as e:
                continue

contagem = Counter(sites_bloqueados)

print("🛡️ Relatório de Firewall - TOP Domínios Bloqueados:\n")
for dominio, freq in contagem.most_common(10):
    print(f"❌ {dominio}: {freq} bloqueio(s) registrado(s)")
